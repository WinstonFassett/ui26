import { Separator as SeparatorReact } from "@registry/separator/separator.tsx";
import SeparatorSvelte from "@registry/separator/separator.svelte";
import SeparatorAstro from "@registry/separator/separator.astro";

Same `Separator` rendered three times per row — React, Svelte, Astro — all sharing `Separator.tv.ts`. Decorative by default; pass `decorative={false}` for the proper `role="separator"` + `aria-orientation` pairing.

## Under the hood

- **Shared config:** `apps/docs/registry/Separator/Separator.tv.ts` — one `orientation` variant (`horizontal` | `vertical`), default `horizontal`.
- **Wrappers:** `apps/docs/registry/Separator/Separator.{tsx,svelte,astro}` — `<div>` with `role="none"` by default, flipped to `role="separator"` + `aria-orientation` when `decorative={false}` is passed.
